function initialsOf(name) {
  return name
    .split(" ")
    .map((p) => p[0])
    .join("")
    .slice(0, 2)
    .toUpperCase();
}

/* ---------- Shared atoms ---------- */
function EditableInput({ value, onChange, placeholder, style }) {
  const [focused, setFocused] = useState(false);
  return (
    <input
      value={value}
      onChange={(e) => onChange(e.target.value)}
      placeholder={placeholder}
      onFocus={() => setFocused(true)}
      onBlur={() => setFocused(false)}
      style={{
        border: `1px solid ${focused ? COLORS.indigo : "transparent"}`,
        borderRadius: 8,
        padding: "3px 7px",
        margin: "-3px -7px",
        background: focused ? "#fff" : "transparent",
        fontFamily: "inherit",
        outline: "none",
        width: "100%",
        boxSizing: "border-box",
        ...style,
      }}
    />
  );
}

/* ---------- Deliverable submission rendering ---------- */
function SubmissionView({ type, submission }) {
  if (!submission) {
    return (
      <div
        style={{
          border: `1px dashed ${COLORS.border}`,
          borderRadius: 12,
          padding: "12px 14px",
          fontSize: 13.5,
          fontWeight: 600,
          color: COLORS.faint,
          fontStyle: "italic",
        }}
      >
        Not submitted yet.
      </div>
    );
  }

  if (type === "text") {
    return (
      <div
        style={{
          background: "#fbfbfd",
          border: `1px solid ${COLORS.border}`,
          borderRadius: 12,
          padding: "12px 14px",
          fontSize: 14,
          color: COLORS.text,
          lineHeight: 1.55,
        }}
      >
        {submission.text}
      </div>
    );
  }

  if (type === "table") {
    return (
      <div style={{ border: `1px solid ${COLORS.border}`, borderRadius: 12, overflow: "hidden" }}>
        <table style={{ width: "100%", borderCollapse: "collapse" }}>
          <thead>
            <tr>
              {submission.columns.map((c, i) => (
                <th
                  key={i}
                  style={{
                    textAlign: "left",
                    fontSize: 12,
                    fontWeight: 800,
                    color: COLORS.indigo,
                    background: COLORS.indigoSoft,
                    padding: "8px 12px",
                    borderBottom: `1px solid ${COLORS.border}`,
                  }}
                >
                  {c}
                </th>
              ))}
            </tr>
          </thead>
          <tbody>
            {submission.rows.map((row, ri) => (
              <tr key={ri}>
                {row.map((cell, ci) => (
                  <td
                    key={ci}
                    style={{
                      fontSize: 13.5,
                      color: COLORS.text,
                      padding: "8px 12px",
                      borderBottom: ri === submission.rows.length - 1 ? "none" : `1px solid ${COLORS.border}`,
                    }}
                  >
                    {cell}
                  </td>
                ))}
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    );
  }

  if (type === "checklist") {
    return (
      <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>
        {submission.items.map((item, i) => (
          <div
            key={i}
            style={{
              display: "flex",
              alignItems: "center",
              gap: 9,
              border: `1px solid ${COLORS.border}`,
              borderRadius: 10,
              padding: "8px 12px",
              background: item.checked ? COLORS.indigoSoft : "#fff",
            }}
          >
            {item.checked ? (
              <CheckSquare size={15} color={COLORS.indigo} strokeWidth={2.5} />
            ) : (
              <Square size={15} color={COLORS.faint} strokeWidth={2.25} />
            )}
            <span style={{ fontSize: 13.5, fontWeight: 600, color: item.checked ? COLORS.text : COLORS.sub }}>
              {item.label}
            </span>
          </div>
        ))}
      </div>
    );
  }

  if (type === "file") {
    return (
      <div
        style={{
          display: "flex",
          alignItems: "center",
          justifyContent: "space-between",
          gap: 12,
          border: `1px solid ${COLORS.border}`,
          borderRadius: 12,
          padding: "10px 14px",
          background: "#fbfbfd",
        }}
      >
        <div style={{ display: "flex", alignItems: "center", gap: 9 }}>
          <FileText size={15} color={COLORS.indigo} />
          <span style={{ fontSize: 13.5, fontWeight: 700, color: COLORS.text }}>{submission.fileName}</span>
        </div>
        <Download size={15} color={COLORS.indigo} />
      </div>
    );
  }

  return null;
}

function EditableTextarea({ value, onChange, placeholder, style, rows = 3 }) {
  const [focused, setFocused] = useState(false);
  return (
    <textarea
      value={value}
      onChange={(e) => onChange(e.target.value)}
      placeholder={placeholder}
      rows={rows}
      onFocus={() => setFocused(true)}
      onBlur={() => setFocused(false)}
      style={{
        border: `1px solid ${focused ? COLORS.indigo : "transparent"}`,
        borderRadius: 8,
        padding: "6px 8px",
        margin: "-6px -8px",
        background: focused ? "#fff" : "transparent",
        fontFamily: "inherit",
        outline: "none",
        width: "100%",
        boxSizing: "border-box",
        resize: "vertical",
        ...style,
      }}
    />
  );
}

function DeliverableCard({ item, onChange }) {
  const typeInfo = DELIVERABLE_TYPES.find((t) => t.id === item.type) || DELIVERABLE_TYPES[0];
  const Icon = typeInfo.icon;
  const hasGrade = item.grade !== null && item.grade !== undefined && item.grade !== "";

  return (
    <div
      style={{
        border: `1px solid ${COLORS.border}`,
        borderRadius: 14,
        padding: 16,
        display: "flex",
        flexDirection: "column",
        gap: 10,
      }}
    >
      <div style={{ display: "flex", alignItems: "center", justifyContent: "space-between", gap: 10 }}>
        <div>
          <span
            style={{
              display: "inline-flex",
              alignItems: "center",
              gap: 6,
              fontSize: 11.5,
              fontWeight: 800,
              color: COLORS.indigo,
              background: COLORS.indigoSoft,
              borderRadius: 999,
              padding: "3px 9px",
              marginBottom: 6,
            }}
          >
            <Icon size={12} />
            {typeInfo.label}
          </span>
          <div style={{ fontSize: 15, fontWeight: 800, color: COLORS.text }}>{item.title}</div>
        </div>
        <div
          style={{
            display: "flex",
            alignItems: "center",
            gap: 6,
            background: hasGrade ? COLORS.indigoSoft : "#f1f3f7",
            borderRadius: 999,
            padding: "4px 6px 4px 12px",
            flexShrink: 0,
          }}
        >
          <EditableInput
            value={item.grade ?? ""}
            onChange={(v) => {
              if (v === "") {
                onChange({ ...item, grade: "" });
                return;
              }
              const digits = v.replace(/[^0-9]/g, "");
              if (digits === "") return;
              const n = Math.min(10, parseInt(digits, 10));
              onChange({ ...item, grade: String(n) });
            }}
            placeholder="—"
            style={{
              fontSize: 14,
              fontWeight: 800,
              color: hasGrade ? COLORS.indigo : COLORS.faint,
              width: 30,
              textAlign: "right",
            }}
          />
        </div>
      </div>

      <SubmissionView type={item.type} submission={item.submission} />

      <div>
        <div style={{ fontSize: 11.5, fontWeight: 800, color: COLORS.faint, textTransform: "uppercase", letterSpacing: 0.4, marginBottom: 6 }}>
          Feedback
        </div>
        <div
          style={{
            background: "#f7f8fb",
            border: `1px solid ${COLORS.border}`,
            borderRadius: 10,
            padding: "9px 11px",
          }}
        >
          <EditableTextarea
            value={item.feedback}
            onChange={(v) => onChange({ ...item, feedback: v })}
            placeholder="Write feedback on this submission..."
            rows={2}
            style={{ fontSize: 13, color: COLORS.text, lineHeight: 1.5, background: "transparent" }}
          />
        </div>
      </div>
    </div>
  );
}

function ChapterSection({ chapter, onChange }) {
  const patchDeliverable = (updated) =>
    onChange({ ...chapter, deliverables: chapter.deliverables.map((d) => (d.id === updated.id ? updated : d)) });

  return (
    <div>
      <div style={{ marginBottom: 10 }}>
        <div style={{ fontSize: 12, fontWeight: 800, color: COLORS.indigo, textTransform: "uppercase", letterSpacing: 0.4 }}>
          {chapter.chapterLabel}
        </div>
        <div style={{ fontSize: 17, fontWeight: 800, color: COLORS.text }}>{chapter.title}</div>
      </div>
      <div style={{ display: "flex", flexDirection: "column", gap: 10 }}>
        {chapter.deliverables.map((d) => (
          <DeliverableCard key={d.id} item={d} onChange={patchDeliverable} />
        ))}
      </div>
    </div>
  );
}

            style={{
              display: "flex",
              alignItems: "center",
              gap: 6,
              fontSize: 12,
              fontWeight: 800,
              color: COLORS.indigo,
              background: COLORS.indigoSoft,
              borderRadius: 999,
              padding: "4px 10px",
            }}
          >
            <Icon size={13} />

            style={{ border: "none", background: "transparent", color: COLORS.faint, cursor: "pointer", padding: 4 }}

            <Trash2 size={16} />

          onChange={(v) => onChange({ ...question, label: v })}

          <div style={{ marginTop: 10, display: "flex", flexDirection: "column", gap: 6 }}>
            {question.options.map((opt, oi) => (
              <div key={oi} style={{ display: "flex", alignItems: "center", gap: 8 }}>
                <span style={{ width: 6, height: 6, borderRadius: 999, background: COLORS.faint }} />

                  value={opt}

                    const options = [...question.options];
                    options[oi] = v;
                    onChange({ ...question, options });
                  }}
                  style={{ fontSize: 14, color: COLORS.sub }}
                />
                <button
                  onClick={() => {
                    const options = question.options.filter((_, i) => i !== oi);
                    onChange({ ...question, options });
                  }}
                  style={{ border: "none", background: "transparent", color: COLORS.faint, cursor: "pointer" }}

                  <Trash2 size={13} />
                </button>
              </div>
            ))}
            <button
              onClick={() => onChange({ ...question, options: [...question.options, "New option"] })}
              style={{
                display: "flex",
                alignItems: "center",
                gap: 6,
                border: "none",
                background: "transparent",
                color: COLORS.indigo,
                fontWeight: 700,
                fontSize: 13,
                cursor: "pointer",
                padding: "4px 0",
                alignSelf: "flex-start",
              }}
            >
              <Plus size={14} /> Add option
            </button>
          </div>
        )}
      </div>
    </div>
  );
}

function FormsTab({ questions, setQuestions }) {
  const update = (id, next) => setQuestions(questions.map((q) => (q.id === id ? next : q)));
  const remove = (id) => setQuestions(questions.filter((q) => q.id !== id));
  const move = (index, dir) => {
    const next = [...questions];
    const [item] = next.splice(index, 1);
    next.splice(index + dir, 0, item);
    setQuestions(next);
  };
  const addQuestion = () =>

  );
}

function StudentsTab({ questions, students, setStudents }) {
  const [filter, setFilter] = useState("all");
  const [search, setSearch] = useState("");
  const [openId, setOpenId] = useState(null);

  const openStudent = students.find((s) => s.id === openId);

  const filtered = students.filter((s) => {
    if (filter !== "all" && s.status !== filter) return false;
    if (search && !s.name.toLowerCase().includes(search.toLowerCase())) return false;
    return true;
  });

  const setStatus = (id, status) => setStudents(students.map((s) => (s.id === id ? { ...s, status } : s)));
  const updateChapters = (id, chapters) => setStudents(students.map((s) => (s.id === id ? { ...s, chapters } : s)));

  if (openStudent) {
    return (
      <StudentDetail
        student={openStudent}
        questions={questions}
        onBack={() => setOpenId(null)}
        onUpdateChapters={(chapters) => updateChapters(openStudent.id, chapters)}
      />
    );
  }

  return (
    <div style={{ maxWidth: 720, margin: "0 auto" }}>
      <div style={{ display: "flex", gap: 10, alignItems: "center", marginBottom: 14, flexWrap: "wrap" }}>
        <div style={{ display: "flex", gap: 6, flexWrap: "wrap" }}>
          <button
            onClick={() => setFilter("all")}
            style={{
              fontSize: 13,
              fontWeight: 800,
              color: filter === "all" ? "#fff" : COLORS.text,
              background: filter === "all" ? COLORS.indigo : "#fff",
              border: `1px solid ${filter === "all" ? COLORS.indigo : COLORS.border}`,
              borderRadius: 999,
              padding: "6px 13px",
              cursor: "pointer",
            }}
          >
            All
          </button>
          {STATUSES.map((s) => {
            const active = filter === s.id;
            return (
              <button
                key={s.id}
                onClick={() => setFilter(s.id)}
                style={{
                  fontSize: 13,
                  fontWeight: 800,
                  color: active ? "#fff" : s.color,
                  background: active ? s.color : s.soft,
                  border: `1px solid ${active ? s.color : s.border}`,
                  borderRadius: 999,
                  padding: "6px 13px",
                  cursor: "pointer",
                }}
              >
                {s.label}
              </button>
            );
          })}
        </div>

        <div style={{ position: "relative", marginLeft: "auto" }}>
          <Search size={14} color={COLORS.faint} style={{ position: "absolute", left: 10, top: 10 }} />
          <input
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            placeholder="Search name"
            style={{
              border: `1px solid ${COLORS.border}`,
              borderRadius: 999,
              padding: "7px 12px 7px 30px",
              fontSize: 13.5,
              outline: "none",
              width: 170,
            }}
          />
        </div>
      </div>

      <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
        {filtered.length === 0 && (
          <div style={{ textAlign: "center", color: COLORS.faint, fontSize: 14, padding: "30px 0" }}>
            No students match this filter.
          </div>
        )}
        {filtered.map((s) => (
          <StudentRow
            key={s.id}
            student={s}
            onOpen={() => setOpenId(s.id)}
            onStatusChange={(status) => setStatus(s.id, status)}
          />
        ))}
      </div>
    </div>
  );
}

function AccManagerTab({ leaders, setLeaders }) {
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");
  const canCreate = name.trim() !== "" && email.trim() !== "";

  const createLeader = () => {
    if (!canCreate) return;
    setLeaders([
      ...leaders,
      { id: nextId("lead"), name: name.trim(), email: email.trim(), createdAt: new Date().toISOString().slice(0, 10) },
    ]);
    setName("");
    setEmail("");
  };

  const inputStyle = {
    width: "100%",
    boxSizing: "border-box",
    border: `1px solid ${COLORS.border}`,
    borderRadius: 10,
    padding: "9px 12px",
    fontSize: 14,
    fontFamily: "inherit",
    outline: "none",
  };

  return (
    <div style={{ maxWidth: 720, margin: "0 auto" }}>
      <div style={{ marginBottom: 18 }}>
        <div style={{ fontSize: 22, fontWeight: 900, color: COLORS.text }}>Student Leaders</div>
        <div style={{ fontSize: 14, color: COLORS.sub, marginTop: 4 }}>
          Enter a name and email, then create the student leader.
        </div>
      </div>

      <div
        style={{
          background: COLORS.card,
          border: `1px solid ${COLORS.border}`,
          borderRadius: 14,
          padding: 16,
          marginBottom: 18,
          display: "flex",
          gap: 10,
          alignItems: "flex-end",
          flexWrap: "wrap",
        }}
      >
        <div style={{ flex: "1 1 200px" }}>
          <div style={{ fontSize: 11.5, fontWeight: 800, color: COLORS.faint, textTransform: "uppercase", letterSpacing: 0.4, marginBottom: 6 }}>
            Name
          </div>
          <input
            value={name}
            onChange={(e) => setName(e.target.value)}
            onKeyDown={(e) => e.key === "Enter" && createLeader()}
            placeholder="Full name"
            style={inputStyle}
          />
        </div>
        <div style={{ flex: "1 1 220px" }}>
          <div style={{ fontSize: 11.5, fontWeight: 800, color: COLORS.faint, textTransform: "uppercase", letterSpacing: 0.4, marginBottom: 6 }}>
            Email
          </div>
          <input
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            onKeyDown={(e) => e.key === "Enter" && createLeader()}
            placeholder="name@tfs.ca"
            style={inputStyle}
          />
        </div>
        <button
          onClick={createLeader}
          disabled={!canCreate}
          style={{
            display: "flex",
            alignItems: "center",
            gap: 7,
            border: "none",
            background: canCreate ? COLORS.indigo : "#d7dbe4",
            color: "#fff",
            fontWeight: 800,
            fontSize: 14,
            borderRadius: 10,
            padding: "10px 16px",
            cursor: canCreate ? "pointer" : "default",
          }}
        >
          <UserPlus size={15} /> Create
        </button>
      </div>

      <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
        {leaders.length === 0 && (
          <div style={{ textAlign: "center", color: COLORS.faint, fontSize: 14, padding: "30px 0" }}>
            No student leaders yet.
          </div>
        )}
        {leaders.map((l) => (
          <LeaderRow key={l.id} leader={l} onRemove={() => setLeaders(leaders.filter((x) => x.id !== l.id))} />
        ))}
      </div>
    </div>
  );
}
