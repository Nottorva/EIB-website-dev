/* Status order follows how a mentor actually moves through outreach:
   contacted -> responsive/unresponsive -> confirmed. Confirmed borrows the
   amber accent already used for mentors elsewhere in the toolset. */
const STATUSES = [
  { id: "contacted", label: "Contacted", color: COLORS.indigo, soft: COLORS.indigoSoft, border: "#dcdcff" },
  { id: "responsive", label: "Responsive", color: COLORS.green, soft: COLORS.greenSoft, border: COLORS.greenBorder },
  { id: "unresponsive", label: "Unresponsive", color: COLORS.red, soft: COLORS.redSoft, border: COLORS.redBorder },
  { id: "confirmed", label: "Confirmed", color: COLORS.amber, soft: COLORS.amberSoft, border: COLORS.amberBorder },
];
const statusOf = (id) => STATUSES.find((s) => s.id === id) || STATUSES[0];

const CONTACT_METHODS = ["Email", "Phone", "Discord", "Other"];
const methodIcon = (method) => {
  if (method === "Email") return Mail;
  if (method === "Phone") return Phone;
  return MessageCircle;
};

let uid = 100;
const nextId = (prefix) => `${prefix}-${uid++}`;

function initialsOf(name) {
  return name
    .split(" ")
    .map((p) => p[0])
    .join("")
    .slice(0, 2)
    .toUpperCase();
}

function makeContact(method, value, preferred, isPublic) {
  return { id: nextId("contact"), method, value, preferred, public: isPublic };
}

/* The single string a student sees in the lesson view's mentor pill,
   derived from whichever contact is marked Public (preferring the one
   that's also marked Preferred, if any). */
function publicContactPreview(mentor) {
  const options = mentor.contacts.filter((c) => c.public && c.value.trim());
  if (options.length === 0) return null;
  const chosen = options.find((c) => c.preferred) || options[0];
  return `${chosen.method} (${chosen.value})`;
}

function ownerOfLesson(mentors, lessonNumber, excludeId) {
  return mentors.find((m) => m.id !== excludeId && m.assignedLessons.includes(lessonNumber));
}

/* ---------- Shared atoms ---------- */
function EditableInput({ value, onChange, placeholder, style }) {
  const [focused, setFocused] = useState(false);
  return (
    <input
      value={value}
      onChange={(e) => onChange(e.target.value)}
      placeholder={placeholder}
      onFocus={() => setFocused(true)}
      onBlur={() => setFocused(false)}
      style={{
        border: `1px solid ${focused ? COLORS.indigo : "transparent"}`,
        borderRadius: 8,
        padding: "3px 7px",
        margin: "-3px -7px",
        background: focused ? "#fff" : "transparent",
        fontFamily: "inherit",
        outline: "none",
        width: "100%",
        boxSizing: "border-box",
        ...style,
      }}
    />
  );
}

function EditableTextarea({ value, onChange, placeholder, style, rows = 3 }) {
  const [focused, setFocused] = useState(false);
  return (
    <textarea
      value={value}
      onChange={(e) => onChange(e.target.value)}
      placeholder={placeholder}
      rows={rows}
      onFocus={() => setFocused(true)}
      onBlur={() => setFocused(false)}
      style={{
        border: `1px solid ${focused ? COLORS.indigo : "transparent"}`,
        borderRadius: 8,
        padding: "6px 8px",
        margin: "-6px -8px",
        background: focused ? "#fff" : "transparent",
        fontFamily: "inherit",
        outline: "none",
        width: "100%",
        boxSizing: "border-box",
        resize: "vertical",
        ...style,
      }}
    />
  );
}

function StatusBadge({ statusId, size = "md" }) {
  const s = statusOf(statusId);
  return (
    <span
      style={{
        display: "inline-flex",
        alignItems: "center",
        fontSize: size === "sm" ? 12 : 13,
        fontWeight: 800,
        color: s.color,
        background: s.soft,
        border: `1px solid ${s.border}`,
        borderRadius: 999,
        padding: size === "sm" ? "3px 9px" : "5px 12px",
      }}
    >
      {s.label}
    </span>
  );
}

function StatusDropdown({ value, onChange }) {
  const s = statusOf(value);
  return (
    <select
      value={value}
      onChange={(e) => onChange(e.target.value)}
      onClick={(e) => e.stopPropagation()}
      style={{
        fontSize: 12.5,
        fontWeight: 800,
        color: s.color,
        background: s.soft,
        border: `1px solid ${s.border}`,
        borderRadius: 999,
        padding: "5px 10px",
        cursor: "pointer",
      }}
    >
      {STATUSES.map((opt) => (
        <option key={opt.id} value={opt.id}>
          {opt.label}
        </option>
      ))}
    </select>
  );
}

/* A small toggle pill for marking a contact Preferred or Public. Both can
   be on at once, that's the common case, not a conflict. */
function TogglePill({ active, onClick, icon: Icon, label, tone = "indigo" }) {
  const toneColor = tone === "amber" ? COLORS.amber : COLORS.indigo;
  const toneSoft = tone === "amber" ? COLORS.amberSoft : COLORS.indigoSoft;
  const toneBorder = tone === "amber" ? COLORS.amberBorder : "#dcdcff";
  return (
    <button
      onClick={onClick}
      style={{
        display: "flex",
        alignItems: "center",
        gap: 5,
        fontSize: 12,
        fontWeight: 800,
        color: active ? toneColor : COLORS.faint,
        background: active ? toneSoft : "#fff",
        border: `1px solid ${active ? toneBorder : COLORS.border}`,
        borderRadius: 999,
        padding: "4px 10px",
        cursor: "pointer",
      }}
    >
      <Icon size={12} strokeWidth={2.5} />
      {label}
    </button>
  );
}

/* ---------- Contact methods editor ---------- */
function ContactRow({ contact, onChange, onDelete, onMakePreferred }) {
  const Icon = methodIcon(contact.method);
  return (
    <div
      style={{
        border: `1px solid ${COLORS.border}`,
        borderRadius: 12,
        padding: "10px 12px",
        display: "flex",
        alignItems: "center",
        gap: 10,
        flexWrap: "wrap",
      }}
    >
      <Icon size={15} color={COLORS.faint} style={{ flexShrink: 0 }} />
      <select
        value={contact.method}
        onChange={(e) => onChange({ ...contact, method: e.target.value })}
        style={{
          fontSize: 12.5,
          fontWeight: 700,
          color: COLORS.sub,
          border: `1px solid ${COLORS.border}`,
          borderRadius: 8,
          padding: "4px 6px",
          background: "#fff",
        }}
      >
        {CONTACT_METHODS.map((m) => (
          <option key={m} value={m}>
            {m}
          </option>
        ))}
      </select>
      <div style={{ flex: 1, minWidth: 140 }}>
        <EditableInput
          value={contact.value}
          onChange={(v) => onChange({ ...contact, value: v })}
          placeholder={contact.method === "Email" ? "name@example.com" : "Contact detail"}
          style={{ fontSize: 14, color: COLORS.text }}
        />
      </div>
      <TogglePill
        active={contact.preferred}
        onClick={onMakePreferred}
        icon={Star}
        label="Preferred"
        tone="indigo"
      />
      <TogglePill
        active={contact.public}
        onClick={() => onChange({ ...contact, public: !contact.public })}
        icon={Eye}
        label="Public"
        tone="amber"
      />
      <button
        onClick={onDelete}
        style={{ border: "none", background: "transparent", color: COLORS.faint, cursor: "pointer", padding: 4 }}
      >
        <Trash2 size={15} />
      </button>
    </div>
  );
}

function ContactsEditor({ contacts, onChange }) {
  const update = (id, next) => onChange(contacts.map((c) => (c.id === id ? next : c)));
  const remove = (id) => onChange(contacts.filter((c) => c.id !== id));
  const makePreferred = (id) => onChange(contacts.map((c) => ({ ...c, preferred: c.id === id })));
  const add = () => onChange([...contacts, makeContact("Email", "", contacts.length === 0, false)]);

  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
      {contacts.length === 0 && (
        <div style={{ fontSize: 13, color: COLORS.faint, fontStyle: "italic", padding: "4px 2px" }}>
          No contact info yet.
        </div>
      )}
      {contacts.map((c) => (
        <ContactRow
          key={c.id}
          contact={c}
          onChange={(next) => update(c.id, next)}
          onDelete={() => remove(c.id)}
          onMakePreferred={() => makePreferred(c.id)}
        />
      ))}
      <button
        onClick={add}
        style={{
          display: "flex",
          alignItems: "center",
          gap: 6,
          border: `1px dashed ${COLORS.border}`,
          background: "#fff",
          color: COLORS.indigo,
          fontWeight: 700,
          fontSize: 13,
          borderRadius: 10,
          padding: "8px 12px",
          cursor: "pointer",
          alignSelf: "flex-start",
        }}
      >
        <Plus size={14} /> Add contact method
      </button>
    </div>
  );
}

function LessonAssignmentsEditor({ mentor, mentors, onToggle }) {
  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
      {MENTOR_LESSONS.map((lesson) => {
        const assigned = mentor.assignedLessons.includes(lesson.number);
        const owner = ownerOfLesson(mentors, lesson.number, mentor.id);
        return (
          <LessonAssignmentRow
            key={lesson.number}
            lesson={lesson}
            assigned={assigned}
            ownerName={owner ? owner.name : null}
            onToggle={() => onToggle(lesson.number, !assigned)}
          />
        );
      })}
    </div>
  );
}

                placeholder="Anything worth remembering about this mentor..."
                rows={3}
                style={{ fontSize: 13.5, color: COLORS.text, lineHeight: 1.5, background: "transparent" }}
              />
            </div>
          </div>

          <div>
            <div style={{ fontSize: 13, fontWeight: 800, color: COLORS.indigo, letterSpacing: 0.4, marginBottom: 10 }}>
              Assigned to lessons
            </div>
            <LessonAssignmentsEditor
              mentor={mentor}
              mentors={mentors}
              onToggle={(lessonNumber, assign) => onToggleLesson(mentor.id, lessonNumber, assign)}
            />
          </div>
        </div>
      ) : (
        <div style={{ display: "flex", flexDirection: "column", gap: 22 }}>
          <div>
            <div style={{ fontSize: 13, fontWeight: 800, color: COLORS.indigo, letterSpacing: 0.4, marginBottom: 10 }}>
              Public role or title
            </div>
            <div style={{ background: COLORS.card, border: `1px solid ${COLORS.border}`, borderRadius: 12, padding: "10px 12px" }}>
              <EditableInput
                value={mentor.publicRole}
                onChange={(v) => onUpdate({ ...mentor, publicRole: v })}
                placeholder="e.g. CEO, Terra CO2 · EIB Panelist"
                style={{ fontSize: 14, color: COLORS.text }}
              />
            </div>
          </div>

          <div>
            <div style={{ fontSize: 13, fontWeight: 800, color: COLORS.indigo, letterSpacing: 0.4, marginBottom: 10 }}>
              Bio shown to students
            </div>
            <div style={{ background: COLORS.card, border: `1px solid ${COLORS.border}`, borderRadius: 12, padding: "10px 12px" }}>
              <EditableTextarea
                value={mentor.publicBio}
                onChange={(v) => onUpdate({ ...mentor, publicBio: v })}
                placeholder="A couple sentences students will read about this mentor..."
                rows={4}
                style={{ fontSize: 14, color: COLORS.text, lineHeight: 1.5 }}
              />
            </div>
          </div>

          <StudentFacingPreview mentor={mentor} />
        </div>
      )}
    </div>
  );
}

/* ---------- List + filter view ---------- */
function MentorsTab({ mentors, setMentors, openId, setOpenId }) {
  const [filter, setFilter] = useState("all");
  const [search, setSearch] = useState("");

  const openMentor = mentors.find((m) => m.id === openId);

  const filtered = mentors.filter((m) => {
    if (filter !== "all" && m.status !== filter) return false;
    if (search && !m.name.toLowerCase().includes(search.toLowerCase())) return false;
    return true;
  });

  const setStatus = (id, status) => setMentors(mentors.map((m) => (m.id === id ? { ...m, status } : m)));
  const updateMentor = (updated) => setMentors(mentors.map((m) => (m.id === updated.id ? updated : m)));
  const removeMentor = (id) => {
    setMentors(mentors.filter((m) => m.id !== id));
    setOpenId(null);
  };
  const toggleLesson = (mentorId, lessonNumber, assign) =>
    setMentors(
      mentors.map((m) => {
        if (m.id === mentorId) {
          const set = new Set(m.assignedLessons);
          if (assign) set.add(lessonNumber);
          else set.delete(lessonNumber);
          return { ...m, assignedLessons: Array.from(set) };
        }
        if (assign) {
          return { ...m, assignedLessons: m.assignedLessons.filter((n) => n !== lessonNumber) };
        }
        return m;
      })
    );

  const addMentor = () => {
    const fresh = {
      id: nextId("mentor"),
      name: "New Mentor",
      status: "contacted",
      contacts: [],
      notes: "",
      lastContactedAt: "",
      assignedLessons: [],
      publicRole: "",
      publicBio: "",
    };
    setMentors([fresh, ...mentors]);
    setOpenId(fresh.id);
  };

  if (openMentor) {
    return (
      <MentorDetail
        mentor={openMentor}
        mentors={mentors}
        onBack={() => setOpenId(null)}
        onUpdate={updateMentor}
        onDelete={() => removeMentor(openMentor.id)}
        onToggleLesson={toggleLesson}
      />
    );
  }

  return (
    <div style={{ maxWidth: 720, margin: "0 auto" }}>
      <div style={{ display: "flex", gap: 10, alignItems: "center", marginBottom: 14, flexWrap: "wrap" }}>
        <div style={{ display: "flex", gap: 6, flexWrap: "wrap" }}>
          <button
            onClick={() => setFilter("all")}
            style={{
              fontSize: 13,
              fontWeight: 800,
              color: filter === "all" ? "#fff" : COLORS.text,
              background: filter === "all" ? COLORS.indigo : "#fff",
              border: `1px solid ${filter === "all" ? COLORS.indigo : COLORS.border}`,
              borderRadius: 999,
              padding: "6px 13px",
              cursor: "pointer",
            }}
          >
            All
          </button>
          {STATUSES.map((s) => {
            const active = filter === s.id;
            return (
              <button
                key={s.id}
                onClick={() => setFilter(s.id)}
                style={{
                  fontSize: 13,
                  fontWeight: 800,
                  color: active ? "#fff" : s.color,
                  background: active ? s.color : s.soft,
                  border: `1px solid ${active ? s.color : s.border}`,
                  borderRadius: 999,
                  padding: "6px 13px",
                  cursor: "pointer",
                }}
              >
                {s.label}
              </button>
            );
          })}
        </div>

        <div style={{ position: "relative", marginLeft: "auto" }}>
          <Search size={14} color={COLORS.faint} style={{ position: "absolute", left: 10, top: 10 }} />
          <input
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            placeholder="Search name"
            style={{
              border: `1px solid ${COLORS.border}`,
              borderRadius: 999,
              padding: "7px 12px 7px 30px",
              fontSize: 13.5,
              outline: "none",
              width: 170,
            }}
          />
        </div>

        <button
          onClick={addMentor}
          style={{
            display: "flex",
            alignItems: "center",
            gap: 6,
            border: "none",
            background: COLORS.indigo,
            color: "#fff",
            fontWeight: 800,
            fontSize: 13,
            borderRadius: 999,
            padding: "7px 14px",
            cursor: "pointer",
          }}
        >
          <Plus size={14} /> Add mentor
        </button>
      </div>

      <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
        {filtered.length === 0 && (
          <div style={{ textAlign: "center", color: COLORS.faint, fontSize: 14, padding: "30px 0" }}>
            No mentors match this filter.
          </div>
        )}
        {filtered.map((m) => (
          <MentorRow key={m.id} mentor={m} onOpen={() => setOpenId(m.id)} onStatusChange={(status) => setStatus(m.id, status)} />
        ))}
      </div>
    </div>
  );
}
